  const BK = {
    APP: 'eoul-lite', FORMAT: 1, STASH: 'eoul-backup:stash', NEXT: 'eoul-backup:stash-next', MAX_BYTES: 5 * 1024 * 1024,
    skip: (k) => typeof k !== 'string' || !k.startsWith('eoul-') || k.startsWith('eoul-backup:') || /:(last|probe)$/.test(k),
    keys() { const out = []; for (let i = 0; i < localStorage.length; i++) { const k = localStorage.key(i); if (!BK.skip(k)) out.push(k); } return out.sort(); },
    snapshot() { const data = {}; for (const k of BK.keys()) { const v = localStorage.getItem(k); if (typeof v === 'string') data[k] = v; } return data; },
    make() { return { app: BK.APP, format: BK.FORMAT, screen: 'd12-hang', at: new Date().toISOString(), data: BK.snapshot() }; },
    stamp() { const d = new Date(); const z = (n) => String(n).padStart(2, '0'); return `${d.getFullYear()}-${z(d.getMonth() + 1)}-${z(d.getDate())}-${z(d.getHours())}${z(d.getMinutes())}`; },
    read(text) {
      if (typeof text !== 'string' || !text.trim()) throw new Error('파일이 비어 있어요.');
      if (text.length > BK.MAX_BYTES) throw new Error('파일이 너무 커요(5MB 초과).');
      let v; try { v = JSON.parse(text); } catch { throw new Error('이 앱의 백업 파일(JSON)이 아니에요.'); }
      if (!v || typeof v !== 'object' || v.app !== BK.APP || v.format !== BK.FORMAT || !v.data || typeof v.data !== 'object' || Array.isArray(v.data)) throw new Error('이 앱의 백업 파일이 아니에요.');
      const data = {};
      for (const [k, val] of Object.entries(v.data)) {
        if (BK.skip(k)) throw new Error(`백업에 모르는 항목이 있어요: ${String(k).slice(0, 40)}`);
        if (typeof val !== 'string') throw new Error(`항목 값이 글자가 아니에요: ${k.slice(0, 40)}`);
        data[k] = val;
      }
      if (!Object.keys(data).length) throw new Error('백업에 자료가 없어요.');
      const engine = data[window.EOUL_MOCK_KEY];
      if (engine !== undefined) {
        // 불러올 때와 같은 엔진 규칙으로 먼저 읽어 본다. 읽지 못할 자료로 바꾸면 앱이 그 기록을 '읽지 못함'으로 멈추기 때문이다.
        let ok;
        if (typeof M !== 'undefined' && typeof M.canReadSaved === 'function') ok = M.canReadSaved(engine);
        else { let st = null; try { st = JSON.parse(engine); } catch {} ok = !!st && st.v === 1 && Array.isArray(st.garments) && Array.isArray(st.journeys); }
        if (!ok) throw new Error('옷장 자료를 이 앱이 읽을 수 없어요. 아무것도 바꾸지 않았어요.');
      }
      return { at: typeof v.at === 'string' ? v.at : '', data };
    },
    count(data) {
      const n = { garments: 0, journeys: 0, threads: 0, reg: 0 };
      try { const st = JSON.parse(data[window.EOUL_MOCK_KEY] || 'null'); if (st) { const trial = new Set(Array.isArray(st.trialIds) ? st.trialIds : []); n.garments = (st.garments || []).filter((g) => !g.trial && !trial.has(g.id)).length; n.journeys = (st.journeys || []).length; } } catch {}
      try { n.threads = (JSON.parse(data[CHAT_KEY] || '{}').threads || []).length; } catch {}
      try { n.reg = Object.keys(JSON.parse(data[REG_KEY] || '{}').items || {}).length; } catch {}
      return n;
    },
    countText: (n) => `내 옷 ${n.garments}벌 · 기록 ${n.journeys}건 · 상담 ${n.threads}개 · 직접 등록 ${n.reg}건`,
    // 복원·되돌리기가 지울 수 있는 키: 이 화면이 쓰는 키와 날씨 지역뿐. 읽지 못한 원문 보관(':unreadable:')·모르는 키는 지우지 않는다.
    removable: (k) => (k.startsWith('eoul-d12-hang:') && !k.includes(':unreadable:')) || k === 'eoul-weather:place',
    // 목표 상태(target)로 바꾼다. 백업에 없는 '지울 수 있는 키'는 지우고, 쓴 뒤 다시 읽어 확인한다.
    // 중간에 막히면 바꾼 키만 원래 값으로 되돌린다. 되돌리기까지 막히면 partial 오류(보관본은 그대로 남는다).
    apply(target) {
      const before = BK.snapshot();
      const drop = Object.keys(before).filter((k) => !(k in target) && BK.removable(k));
      const touched = [];
      try {
        for (const k of drop) { touched.push(k); localStorage.removeItem(k); }
        for (const [k, v] of Object.entries(target)) { if (before[k] === v) continue; touched.push(k); localStorage.setItem(k, v); }
        for (const k of drop) if (localStorage.getItem(k) !== null) throw new Error('verify');
        for (const [k, v] of Object.entries(target)) if (localStorage.getItem(k) !== v) throw new Error('verify');
      } catch {
        try {
          for (const k of touched) if (!(k in before)) localStorage.removeItem(k); // 새로 생긴 키부터 지워 공간을 비운다.
          for (const k of touched) if (k in before) localStorage.setItem(k, before[k]);
          for (const k of touched) if (localStorage.getItem(k) !== (k in before ? before[k] : null)) throw new Error('verify');
        } catch {
          throw Object.assign(new Error("중간에 저장이 막혀 일부만 바뀌었어요. 바꾸기 전 자료는 보관본에 남아 있어요. 저장 공간을 비운 뒤 '복원 전으로 되돌리기'를 눌러 주세요."), { code: 'partial' });
        }
        throw new Error('저장 공간이 부족하거나 저장이 막혀 바꾸지 않았어요. 지금 자료는 그대로예요.');
      }
    },
    // 복원: 현재 값을 보관본(가장 최근 한 벌)으로 먼저 저장·확인한 뒤에만 바꾼다.
    // 보관본 두 칸: 지금 되돌리기 지점(STASH)은 이번 복원이 끝까지 성공하기 전에는 절대 건드리지 않는다.
    // ① 이번 복원 전 상태를 NEXT 칸에 쓰고 확인 → ② 바꾸기 → ③ 성공하면 NEXT를 STASH로 올리고 NEXT를 지운다.
    // 깨끗이 되돌려졌으면 NEXT만 지운다(앞선 STASH 그대로). 일부만 바뀌었으면 NEXT를 복구 지점으로 남긴다.
    // 올리기(③)가 막혀도 NEXT가 남아 있어 되돌리기는 NEXT를 쓴다.
    restore(data) {
      // 앞선 복원의 임시 칸이 남아 있으면(일부만 바뀌었거나 올리기가 막힘) 그것이 유일한 깨끗한 복구 지점일 수 있다.
      // 어느 경우든 새 복원으로 덮지 않도록, 먼저 '복원 전으로 되돌리기'를 하게 하고 이번 복원은 시작하지 않는다.
      let pending = null; try { pending = localStorage.getItem(BK.NEXT); } catch {}
      if (pending !== null) throw Object.assign(new Error("앞선 복원이 끝까지 정리되지 않았어요. 먼저 '복원 전으로 되돌리기'를 한 뒤 다시 복원해 주세요."), { code: 'pending' });
      const before = BK.snapshot();
      const text = JSON.stringify({ at: new Date().toISOString(), data: before });
      const dropNext = () => { try { localStorage.removeItem(BK.NEXT); } catch {} };
      try { localStorage.setItem(BK.NEXT, text); if (localStorage.getItem(BK.NEXT) !== text) throw new Error('verify'); } catch { dropNext(); throw new Error('되돌릴 보관본을 저장하지 못해 복원하지 않았어요. 저장 공간을 확인해 주세요.'); }
      try { BK.apply(data); } catch (e) { if (e.code !== 'partial') dropNext(); throw e; }
      try { localStorage.setItem(BK.STASH, text); if (localStorage.getItem(BK.STASH) !== text) throw new Error('verify'); localStorage.removeItem(BK.NEXT); } catch { /* NEXT가 남아 되돌리기 지점 역할을 한다 */ }
      return before;
    },
    // 되돌리기 지점: 아직 올리지 못했거나 일부만 바뀐 복원의 NEXT가 있으면 그것이 가장 최근 지점이다.
    stash() {
      for (const k of [BK.NEXT, BK.STASH]) {
        try { const st = JSON.parse(localStorage.getItem(k) || 'null'); if (st && st.data && typeof st.data === 'object' && !Array.isArray(st.data)) return { ...st, key: k }; } catch {}
      }
      return null;
    },
    // 되돌리기: 보관본 상태로 같은 방식(확인·부분 되돌림)으로 바꾸고, 끝까지 성공했을 때만 보관본을 지운다.
    undo() {
      const st = BK.stash(); if (!st) throw new Error('되돌릴 보관본이 없어요.');
      const target = {};
      for (const [k, v] of Object.entries(st.data)) if (!BK.skip(k) && typeof v === 'string') target[k] = v;
      BK.apply(target);
      try { localStorage.removeItem(st.key); } catch {}
    },
  };
